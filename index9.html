String in C

What is String ?

In C language, a string is not a separate data type. It is implemented as an array of characters that ends with a special character \0 (null character). Strings are usually enclosed in double quotes (" "), for example "Hello". Functions for string handling are available in the string.h library, such as strlen(), strcpy(), strcat(), and strcmp().

Short Definition :- A string in C is a sequence of characters stored in a character array, terminated by a null character (\0).

Example:-

  • #include stdio.h
  • int main()
  • {
  • char name[6] = "Preet";
  • printf("Name : %s", name);
  • }

Examples:-

▶️ strlen :- strlen() Function use to return the length of the string.

#include stdio.h

#include string.h

int main ()

{

char name[80];

int length;

printf("Enter Name:");

gets(name);

length=strlen(name);

printf("\nYour name has %d characters", length);

}

▶️ strcat :- The strcat() is a string function.Which use to combine two strings strcat(str1,str2).

#include stdio.h

#include string.h

int main ()

{

char first_name[20], second_name[20], fullname[20];

printf("Enter First_Name:");

gets(first_name);

printf("Enter Surname:");

gets(second_name);

fullname = strcat(first_name,second_name);

printf("Your Full Name : %s",fullname);

}

▶️ Strcpy :- In c Langauge, You cannot simply assign one character arry to another like int a=10, b; a=b;

#include stdio.h

#include string.h

int main ()

{

char stu_name[8],emp_name[8];

printf("Enter Name:");

gets(stu_name);

strcpy(emp_name, stu_name);

printf("Student Name : %s \n Employee Name : %s", stu_name, emp_name);

}

▶️ strlwr and strupr case

#include stdio.h

#include string.h

int main ()

{

char name[20];

printf("Enter Name : ");

gets(name);

strupr(name);

printf("Name in Uppercase: %s\n", name);

strlwr(name);

printf("Name in Lowercase: %s", name);

}

▶️ Reverse String Function (strrev())

#include stdio.h

#include string.h

int main ()

{

char name[20];

printf("Enter Name : ");

gets(name);

strrev(name);

printf("Reverse Name : %s", name);

}

▶️ strset :- Replace string to One Character for using strset() Function.

#include stdio.h

#include string.h

int main ()

{

char str[10]="Abhi";
